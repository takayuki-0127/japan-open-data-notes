```js
// https://www.e-stat.go.jp/ を開いた状態で、開発者ツールのコンソールで実行
const csv = {s01:'000040472270', s02:'000040472271', s04:'000040472273', i01:'000040473360', i10:'000040473369',
  i22:'000040473381', i24:'000040473383', i26:'000040473385', i36:'000040473395', i39:'000040473398'};
const xl = {c_ts_size:'000001086172', c_ts_fam:'000012671095', c_ts_famage:'000012671096', c25_11_1:'000040506463'};
const o = {fetched: new Date().toISOString(), csv_ids: csv, xlsx_ids: xl};
for (const [k, id] of Object.entries(csv)) {
  const b = await (await fetch('/stat-search/file-download?statInfId=' + id + '&fileKind=1')).arrayBuffer();
  o[k] = new TextDecoder('shift_jis').decode(b); await new Promise(r => setTimeout(r, 1500)); }
for (const [k, id] of Object.entries(xl)) {
  const b = new Uint8Array(await (await fetch('/stat-search/file-download?statInfId=' + id + '&fileKind=0')).arrayBuffer());
  let s = ''; for (let i = 0; i < b.length; i += 8192) s += String.fromCharCode.apply(null, b.subarray(i, i + 8192));
  o[k] = btoa(s); await new Promise(r => setTimeout(r, 1500)); }
const a = document.createElement('a'); a.href = URL.createObjectURL(new Blob([JSON.stringify(o)], {type: 'application/json'}));
a.download = 'f117_raw.json'; a.click();
```

JSON から CSV・xlsx を取り出すには（同梱済み）：

```python
import json, base64
o = json.load(open('data/raw/f117_raw.json'))
for k, v in o.items():
    if k in ('fetched', 'csv_ids', 'xlsx_ids'): continue
    if k.startswith('c'): open(f'data/raw/{k}.xlsx', 'wb').write(base64.b64decode(v))
    else: open(f'data/raw/{k}.csv', 'w', encoding='utf-8').write(v)
```
